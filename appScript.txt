function doGet(e) {
  return doPost(e);
}

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var data = {};

    if (e.postData && e.postData.contents) {
      try {
        data = JSON.parse(e.postData.contents);
      } catch (err) {
        data = e.parameter || {};
      }
    } else {
      data = e.parameter || {};
    }

    var timestamp     = sanitizeField(data.date || new Date().toLocaleString("en-IN", { timeZone: "Asia/Kolkata" }));
    var serviceType   = sanitizeField(data.type || data.form_type || "General Inquiry");
    var name          = sanitizeField(data.name || "Anonymous");
    var contact       = sanitizeField(data.contact || data.phone || "N/A");
    var email         = sanitizeField(data.email || "N/A");
    var dob           = sanitizeField(data.dob || "N/A");
    var projectSize   = sanitizeField(data.project_size || "N/A");
    var projectBudget = sanitizeField(data.project_budget || "N/A");
    var location      = sanitizeField(data.location || "N/A");
    var challenges    = sanitizeField(data.challenges || data.message || data.details || "N/A");

    // Single Tab: All Leads Summary
    var sheet = getOrCreateSheet(ss, "All Leads Summary", [
      "Timestamp",
      "Service / Form Name",
      "Name",
      "Contact No",
      "Email",
      "Date of Birth",
      "Project Size",
      "Budget",
      "Location",
      "Challenges / Message / Notes"
    ]);

    sheet.appendRow([
      timestamp,
      serviceType,
      name,
      contact,
      email,
      dob,
      projectSize,
      projectBudget,
      location,
      challenges
    ]);

    return ContentService
      .createTextOutput(JSON.stringify({ status: "success", message: "Lead recorded successfully!" }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: "error", message: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function sanitizeField(val) {
  if (val === null || val === undefined) return "N/A";
  var str = String(val).trim();
  if (str === "") return "N/A";
  // Prepend single quote if string starts with + or = to force Google Sheets plain text format
  if (str.indexOf("+") === 0 || str.indexOf("=") === 0) {
    return "'" + str;
  }
  return str;
}

function getOrCreateSheet(ss, sheetName, headers) {
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
  }
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(headers);
    sheet.getRange(1, 1, 1, headers.length)
      .setFontWeight("bold")
      .setBackground("#EAE1D5")
      .setColor("#2A241F")
      .setHorizontalAlignment("center");
    sheet.setFrozenRows(1);
  }
  return sheet;
}
